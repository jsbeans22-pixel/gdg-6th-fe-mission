# 자바스크립트의 문법
(파이썬으로 기본 문법들을 익혔기에 차이점 위주로 기술)

## 변수 선언
let name = "APPLE";   
const num = 20;
let은 변경 가능, const는 변경 불가능
파이썬처럼 동적 타입 언어라서 따로 자료형 정의하지 않는다

## 반복문
while () {}
>>파이썬이랑 다르게 ()안에 조건을 넣고 {}안에 실행할 문장을 넣는다, 조건 뒤에 : 없음

for (let i = 0; i < 5; i++) {}
>>in range를 쓰는 파이썬보다는 c언어와 비슷함

## 조건문
if () {}
>>() {} 존재, : 없음
## 함수
function으로 정의, ()안에 변수

## 배열
파이썬의 리스트와 거의 비슷, 웹페이지 제작에서 중요하다고 한다
for (let num of numbers) {
  console.log(num);
}
파이썬처럼 배열 반복으로 하나씩 값 확인 가능
